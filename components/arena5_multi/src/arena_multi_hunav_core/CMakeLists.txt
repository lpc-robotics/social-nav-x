cmake_minimum_required(VERSION 3.8)
project(arena_multi_hunav_core)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(hunav_msgs REQUIRED)
find_package(arena_multi_hunav_msgs REQUIRED)
add_library(multi_sfm src/core.cpp)
target_include_directories(multi_sfm PUBLIC include PRIVATE vendor/lightsfm/include)
ament_target_dependencies(multi_sfm rclcpp hunav_msgs arena_multi_hunav_msgs)
add_executable(multi_sfm_server src/server.cpp)
add_executable(multi_sfm_benchmark src/benchmark.cpp)
target_link_libraries(multi_sfm_benchmark multi_sfm)
target_link_libraries(multi_sfm_server multi_sfm)
ament_target_dependencies(multi_sfm_server rclcpp arena_multi_hunav_msgs)
install(TARGETS multi_sfm multi_sfm_server multi_sfm_benchmark ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY vendor DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_core test/test_core.cpp)
  # Conda Humble's older gtest omits this header with the current compiler.
  if(TARGET gtest)
    target_compile_options(gtest PRIVATE -include cstdint)
  endif()
  target_link_libraries(test_core multi_sfm)
endif()
ament_package()
